import os
import re

companies = [
    "600519", "000333", "300750", "002594", "600900", 
    "000858", "600309", "000651", "600276", "002415", "601012"
]

def get_context(lines, keyword, num_lines=2):
    for i, line in enumerate(lines):
        if keyword in line:
            return "\n".join(lines[i:i+num_lines+1])
    return None

for code in companies:
    filename = f"{code}.txt"
    if not os.path.exists(filename):
        continue
        
    with open(filename, 'r', encoding='utf-8') as f:
        content = f.read()
        
    print(f"--- {code} ---")
    
    # Find Income Statement
    idx_income = content.find("合并利润表")
    if idx_income == -1:
        idx_income = content.find("利润表")
        
    if idx_income != -1:
        snippet = content[idx_income:idx_income+3000]
        lines = snippet.split('\n')
        unit_line = get_context(lines, "单位", 0)
        print("Income Unit:", unit_line)
        
        rev_context = get_context(lines, "营业收入", 2)
        if not rev_context:
            rev_context = get_context(lines, "营业总收入", 2)
        print("Income Rev:\n" + str(rev_context))
    else:
        print("Income Statement not found")
        
    # Find Cash Flow Statement
    idx_cash = content.find("合并现金流量表")
    if idx_cash == -1:
        idx_cash = content.find("现金流量表")
        
    if idx_cash != -1:
        snippet = content[idx_cash:idx_cash+4000]
        lines = snippet.split('\n')
        unit_line = get_context(lines, "单位", 0)
        print("Cash Unit:", unit_line)
        
        ocf_context = get_context(lines, "经营活动产生的现金流量净额", 2)
        print("Cash OCF:\n" + str(ocf_context))
        
        capex_context = get_context(lines, "购建固定资产", 2)
        print("Cash CAPEX:\n" + str(capex_context))
    else:
        print("Cash Flow Statement not found")
    print()
